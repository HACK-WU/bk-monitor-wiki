```python
# issue_processor.py
class IssueAggregationProcessor:
    def process(self, alert, strategy) -> bool:
        # 1. 配置校验
        if not self._check_config(alert, strategy):
            return False
        
        # 2. 指纹计算
        fingerprint = gen_issue_fingerprint(
            strategy.id,
            strategy.issue_config.aggregate_dimensions,
            alert.origin_alarm.data.dimensions,
        )
        if not fingerprint:
            return False
        
        # 3. 查找活跃 Issue（三级）
        issue = self._find_active_issue(fingerprint)
        
        if issue:
            # 4a. 关联已有 Issue
            self._associate_alert(issue)
        else:
            # 4b. 创建新 Issue（分布式锁保护）
            lock = self._acquire_lock(fingerprint)
            if not lock:
                return False
            issue = self._create_issue(config, fingerprint, dim_values)
            self._associate_alert(issue)
            self._maybe_dispatch_llm_title(issue)
        
        return True

def gen_issue_fingerprint(strategy_id, aggregate_dimensions, data_dimensions):
    """指纹计算：count_md5(strategy_id + sorted dims)"""
    payload = [f"strategy:{strategy_id}"]
    for key in sorted(aggregate_dimensions):
        value = data_dimensions.get(key)
        if value is None or str(value).strip() == "":
            return None
        payload.append(f"{key}={str(value).strip()}")
    return count_md5(payload)
```

## 持久化

```python
def _persist_and_cache(self, active: bool):
    """双写：先 ES 后 Redis"""
    # ES UPSERT（重试 1 次）
    try:
        self.save()
    except Exception:
        self.save()  # 重试 1 次
    
    # Redis 缓存
    if active:
        self._update_redis_cache()
    else:
        self._delete_redis_cache()
```
