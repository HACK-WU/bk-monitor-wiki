## 用户态授权

```python
# views.py
class TAPDAuthPermission(BasePermission):
    """前置校验 Redis tapd_uat:{tenant}:{user} token"""
    def has_permission(self, request, view):
        token = redis_client.get(f"tapd_uat:{tenant}:{user}")
        if not token:
            if request.data.get("success_url"):
                return Response({"auth_url": build_auth_url()}, status=403)
            return Response({"message": "请先完成 TAPD 授权"}, status=403)
        return True
```

```python
# views.py
@csrf_exempt
def tapd_oauth_callback(request):
    """用户态授权码兑换，写回 tapd_uat token"""
    code = request.GET.get("code")
    token = exchange_code_for_token(code)
    redis_client.set(f"tapd_uat:{tenant}:{user}", token)
    return redirect(success_url)
```

## 创建 TAPD 单

```python
# resources.py
class CreateTapdResource(Resource):
    def perform_request(self, validated_request_data):
        # 1. 调用 TAPD API 创建工单
        tapd_item = tapd_client.create_item(
            workspace_id=workspace_id,
            item_type=item_type,  # story/bug/task
            fields=fields,
        )
        
        # 2. 持久化关联
        IssueTapdRelation.objects.create(
            issue_id=issue_id,
            tapd_workspace_id=workspace_id,
            tapd_item_id=tapd_item.id,
            tapd_item_type=item_type,
        )
        
        # 3. 记录活动日志
        issue.add_activity(
            activity_type="create_tapd",
            content=json.dumps(tapd_item),
            operator=operator,
        )
        
        return tapd_item
```

## 关联已有 TAPD 单

```python
class LinkIssueToTapdResource(Resource):
    def perform_request(self, validated_request_data):
        # 1. 批量查重
        existing = IssueTapdRelation.objects.filter(
            issue_id=issue_id,
            tapd_item_id__in=item_ids,
        ).values_list("tapd_item_id", flat=True)
        
        new_items = [i for i in item_ids if i not in existing]
        
        # 2. 批量创建关联
        IssueTapdRelation.objects.bulk_create([
            IssueTapdRelation(issue_id=issue_id, tapd_item_id=iid)
            for iid in new_items
        ])
        
        # 3. 记录活动日志
        issue.add_activity(
            activity_type="tapd_link",
            content=json.dumps({"linked_items": new_items}),
            operator=operator,
        )
```

## 工作区解绑/重绑

```python
class UnbindTapdWorkspaceResource(Resource):
    def perform_request(self, validated_request_data):
        """持久化 tombstone，阻断周期任务自动回绑"""
        TapdWorkspaceManualUnbind.objects.create(
            bk_biz_id=bk_biz_id,
            tapd_workspace_id=workspace_id,
        )

class RebindTapdWorkspaceResource(Resource):
    def perform_request(self, validated_request_data):
        """清除 tombstone，恢复自动同步"""
        TapdWorkspaceManualUnbind.objects.filter(
            bk_biz_id=bk_biz_id,
            tapd_workspace_id=workspace_id,
        ).delete()
```
