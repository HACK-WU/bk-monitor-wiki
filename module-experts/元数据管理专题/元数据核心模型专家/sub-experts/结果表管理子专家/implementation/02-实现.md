## 二、modify 逐项更新流程

**文件**: `result_table.py` 行 1149-1500

### 2.1 流程概览

```
modify() 逐项检查并更新:
1. table_name_zh → 直接赋值
2. default_storage → 校验存储存在 → 赋值
3. field_list（仅 free schema）:
   a. 删除旧用户配置字段
   b. 删除旧清洗类 option
   c. 重建默认字段
   d. 批量创建新字段
   e. 级联更新 CMDB 拆分表
4. label → 校验存在性 → 赋值
5. external_storage → 逐项创建/更新
6. need_delete_storages → 逐项删除
7. option → 清理清洗类 → 批量写入
8. is_enable → 启用时创建 ES 索引 → 停用时删除数据链路
9. bk_biz_id_alias → 直接赋值
10. data_label → 直接赋值
11. labels → 直接赋值
12. save() → 推送空间路由
13. apply_datalink() 或 delete_datalink()
```

## 三、upgrade_result_table 实现

**文件**: `result_table.py` 行 1505-1570

```python
def upgrade_result_table(self, operator):
    # 1. 校验：非全业务且未删除
    if self.bk_biz_id == 0 or self.is_deleted:
        raise ValueError(_("结果表不可操作，请确认后重试"))
    
    # 2. 标记旧记录
    self.is_deleted = True
    self.bk_biz_id = 0
    self.last_modify_user = operator
    self.last_modify_time = datetime.datetime.now()
    self.save()  # 必须在改 table_id 前 save
    
    # 3. 提取新 table_id
    new_table_id = re.match(r"\d+_(?P<table_id>(\w|_|\.)+)", self.table_id).group("table_id")
    
    # 4. 字段迁移
    ResultTableField.objects.filter(
        table_id=self.table_id, bk_tenant_id=self.bk_tenant_id
    ).update(table_id=new_table_id)
    
    # 5. 存储迁移（深拷贝）
    for storage_str in list(self.REAL_STORAGE_DICT.keys()):
        storage_config = self.REAL_STORAGE_DICT[storage_str]
        try:
            storage = storage_config.objects.get(table_id=self.table_id, bk_tenant_id=self.bk_tenant_id)
        except storage_config.DoesNotExist:
            continue
        new_storage = copy.deepcopy(storage)
        storage.delete()
        new_storage.table_id = new_table_id
        new_storage.bk_tenant_id = self.bk_tenant_id
        new_storage.save()
    
    # 6. 数据源关系迁移
    DataSourceResultTable.objects.filter(
        table_id=self.table_id, bk_tenant_id=self.bk_tenant_id
    ).update(table_id=new_table_id)
    
    # 7. 自身 table_id 更新
    self.table_id = new_table_id
    self.save()
    
    return True
```

```python
def apply_datalink(self, force_update: bool = False) -> None:
    datasource = self.get_related_datasource()
    target_bk_biz_id = self.get_target_bk_biz_id()
    
    # 获取结果表选项
    options = {
        option.name: option.get_value()
        for option in ResultTableOption.objects.filter(
            bk_tenant_id=self.bk_tenant_id, table_id=self.table_id
        )
    }
    
    if self.default_storage == ClusterInfo.TYPE_INFLUXDB:
        # V4 链路条件
        is_v4_datalink_etl_config = (
            datasource.created_from == DataIdCreatedFromSystem.BKDATA.value
            and datasource.etl_config in ENABLE_V4_DATALINK_ETL_CONFIGS
        )
        
        # 插件 V4 链路
        is_plugin_v4_datalink = options.get(
            ResultTableOption.OPTION_ENABLE_PLUGIN_V4_DATA_LINK, False
        ) and datasource.etl_config in [EtlConfigs.BK_EXPORTER.value, EtlConfigs.BK_STANDARD.value]
        
        if is_plugin_v4_datalink and not is_v4_datalink_etl_config:
            is_v4_datalink_etl_config = True
        
        if (is_v4_datalink_etl_config and settings.metadata.enable_v2_vm_data_link) \
           or not settings.metadata.enable_influxdb_storage:
            
            # 插件类型：先注册到 BkBase
            if is_plugin_v4_datalink and datasource.created_from == DataIdCreatedFromSystem.BKGSE.value:
                datasource.register_to_bkbase(bk_biz_id=target_bk_biz_id, namespace=BKBASE_NAMESPACE_BK_MONITOR)
            
            # 异步任务
            on_commit(
                func=lambda: access_bkdata_vm.delay(
                    bk_tenant_id, target_bk_biz_id, table_id, bk_data_id,
                    is_v4_datalink_etl_config, force_update=force_update,
                ),
                using=config.DATABASE_CONNECTION_NAME,
            )
    
    elif self.default_storage in [ClusterInfo.TYPE_ES, ClusterInfo.TYPE_DORIS]:
        # 日志 V4 链路
        if options and options.get(ResultTableOption.OPTION_ENABLE_V4_LOG_DATA_LINK, False):
            apply_log_datalink(bk_tenant_id=self.bk_tenant_id, table_id=self.table_id)
        # 事件组 V4 链路
        elif datasource.etl_config == EtlConfigs.BK_STANDARD_V2_EVENT.value:
            apply_event_group_datalink(bk_tenant_id=self.bk_tenant_id, table_id=self.table_id)
    
    # BKGSE 来源：刷新 Consul
    datasource = self.get_related_datasource(refresh=True)
    if datasource.created_from == DataIdCreatedFromSystem.BKGSE.value:
        datasource.refresh_consul_config()
```

```python
def set_metric_split(self, cmdb_level, operator):
    # 1. 校验：非公共数据源
    if not (self.data_source.bk_data_id >= 10000 or allow_all_cmdb_level):
        raise ValueError("公共数据源不支持 CMDB 层级拆分")
    
    # 2. 检查是否已存在拆分记录
    if CMDBLevelRecord.enable_object.filter(
        source_table_id=self.table_id, cmdb_level=cmdb_level, bk_tenant_id=self.bk_tenant_id
    ).exists():
        raise ValueError("已存在相同层级的拆分记录")
    
    # 3. 检查自身是否已是拆分目标
    if CMDBLevelRecord.enable_object.filter(
        target_table_id=self.table_id, bk_tenant_id=self.bk_tenant_id
    ).exists():
        raise ValueError("当前结果表已是拆分目标")
    
    # 4. 获取或创建输出数据源
    # 5. 创建 CMDBLevelRecord
    record = CMDBLevelRecord.objects.create(
        source_table_id=self.table_id,
        target_table_id=target_table_id,
        cmdb_level=cmdb_level,
        bk_tenant_id=self.bk_tenant_id,
        is_enable=True,
    )
    return record
```
