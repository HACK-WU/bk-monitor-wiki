## 二、StorageResultTable 接口（Mixin）

```python
class StorageResultTable:
    STORAGE_TYPE: str = None          # 存储类型常量
    UPGRADE_FIELD_CONFIG: tuple = ()  # 可升级字段列表
    JSON_FIELDS: tuple = ()           # JSON 类型字段列表

    # === 抽象方法 ===
    @classmethod
    @abc.abstractmethod
    def create_table(cls, table_id, is_sync_db=False, **kwargs) -> Self

    @abc.abstractmethod
    def get_client(self) -> Any

    @abc.abstractmethod
    def add_field(self, field) -> None

    # === 实例方法 ===
    def consul_config(self) -> dict | None

    def update_storage(self, **kwargs) -> bool

    # === 属性 ===
    @property
    def storage_type(self) -> str

    @property
    def storage_cluster(self) -> ClusterInfo
```
